function mostrarCatalogo(items) { let contenedor = document.getElementById('contenedorCatalogo'); contenedor.innerHTML = ''; let categorias = ["Películas", "Series", "Anime", "Telenovelas"]; categorias.forEach(cat => { let filtrados = items.filter(i => i.categoria === cat); if(filtrados.length > 0) { let label = document.createElement('div'); label.className = 'category-label'; label.innerText = cat; contenedor.appendChild(label); let grid = document.createElement('div'); grid.className = 'media-grid'; filtrados.forEach(item => { let card = document.createElement('a'); card.className = 'media-card'; card.href = `ver.html?id=${item.id}`; // <--- AQUÍ ESTÁ LA MAGIA QUE CONECTA CON EL REPRODUCTOR card.innerHTML = `
${item.titulo}
`; grid.appendChild(card); }); contenedor.appendChild(grid); } }); }
